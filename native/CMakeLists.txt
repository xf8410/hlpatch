cmake_minimum_required(VERSION 3.18)
project(umahook VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(DOBBY_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/Dobby")
file(GLOB_RECURSE DOBBY_SRC "${DOBBY_DIR}/source/*.cpp" "${DOBBY_DIR}/source/*.cc")
add_library(main SHARED src/main.cpp src/il2cpp_hook.cpp src/game_hooks.cpp src/http_server.cpp ${DOBBY_SRC})
target_include_directories(main PRIVATE src include ${DOBBY_DIR}/include ${DOBBY_DIR}/source)
target_link_libraries(main log dl)
target_compile_options(main PRIVATE -Wall -Wno-unused-parameter -fno-rtti -fno-exceptions -fvisibility=hidden)
target_link_options(main PRIVATE -Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/export.map)
set_target_properties(main PROPERTIES OUTPUT_NAME "main" PREFIX "lib")
